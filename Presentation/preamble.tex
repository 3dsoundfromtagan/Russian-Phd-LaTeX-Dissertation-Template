\section{Актуальность и мотивация}

\begin{frame}[plain, noframenumbering]
	\frametitle{Актуальность}
	\begin{center}
		Излучение среднего ИК--диапазона (3--50)~мкм
	\end{center}
	
	\begin{columns}[t]
		\begin{column}{0.59\textwidth}
			\textbf{Применение}:
			\begin{itemize}
				\item Пики поглощения воды $\Rightarrow$ медицина (хирургия, дерматология)
				\item Пики поглощения молекулярных связей $\Rightarrow$ спектроскопия				
			\end{itemize}			
			
			\textbf{Метод:} генерация разностной частоты (ГРЧ) в кристалле с регулярной доменной структурой (PPLN).
			
			\textbf{Накачка:} синхронизированные импульсы волоконных лазеров ($\sim\!1$ мкм и $\sim\!1.5$ мкм) в одном волокне.
			
			\textbf{Преимущества:} компактность, волоконный инструмент, перестройка длины волны, узкая линия излучения.
			
		\end{column}
		\begin{column}{0.4\textwidth}
			\fixme{спереть картинку у Игоря}
			\includegraphics[width=\linewidth]{Presentation/images/ppln}
		\end{column}
	\end{columns}
\end{frame}


\begin{frame}[plain, noframenumbering]
	\frametitle{Актуальность}
	Генерация излучения с длиной волны 3 мкм в н-о кристалле PPLN
	\includegraphics[width=\linewidth]{Presentation/images/problem}
	\begin{columns}[t]
		\begin{column}{0.59\textwidth}
			Н-о эффекты в волокне:
			\begin{itemize}
				\item Высокая пиковая мощность оптических импульсов
				\item Взаимное влияние импульсов на двух длинах волн
				\item Практичность $\Rightarrow$ большая длина волокна доставки
				\item Следствие: \textbf{нелинейные эффекты второго и третьего порядка}
				
			\end{itemize}
			
		\end{column}
		\begin{column}{0.4\textwidth}
			Тепловые эффекты в н-о кристалле:
			\begin{itemize}
				\item Разогрев за счёт оптического поглощения $\Rightarrow$ \textbf{необходимость его прецизионного определения				}
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}
