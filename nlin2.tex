			\end{itemize}

	\frametitle{Модель}
	
	Динамика ГВГ в ОВ\footnote{\scriptsize Антонюк Б. П., Антонюк В. Б. Самоорганизация возбуждений в германосиликатных волоконных световодах и ее роль в генерации второй гармоники //Успехи физических наук. – 2001. – Т. 171. – №. 1. – С. 61-78.}:
	
	\vspace{0.2cm}
	
	\begin{columns}
		\begin{column}{0.57\textwidth}	
		\vspace{-0.6cm}
			\only<1>{
				\begin{equation*}
					\begin{aligned}
						\frac{\partial A_1}{\partial z} &= \im \frac{4 \pi \omega \chi^{\left(3\right)}}{c n_{\omega}}\left( A_0^{\ast} A_1^{\ast} A_2 + \gamma_{11} A_1 A_1^{\ast} A_1 + 2\gamma_{12} A_2 A_2^{\ast} A_1\right) \\
						\frac{\partial A_2}{\partial z} &= \im \frac{4 \pi \omega \chi^{\left(3\right)}}{c n_{2\omega}}\left(A_0 A_1^{2}  + 2\gamma_{12} A_1 A_1^{\ast} A_2 + \gamma_{22} A_2 A_2^{*} A_2 \right) \\
						\frac{\partial A_{0}}{\partial t} &= \alpha_2 \left|A_2\right|^2 \cdot \left(u e^{\im\left[\arg{A_2}-2\arg{A_1}\right]}-A_{0}\right)
					\end{aligned}
				\end{equation*}
			}	
			
			\only<2->{
				\begin{equation*}
					\begin{aligned}
						\frac{\partial A_1}{\partial z} &= \im \frac{4 \pi \omega \chi^{\left(3\right)}}{c n_{\omega}}\left( A_0^{\ast} A_1^{\ast} A_2 + \highlight{\gamma_{11} A_1 A_1^{\ast} A_1 + 2\gamma_{12} A_2 A_2^{\ast} A_1}\right) \\
						\frac{\partial A_2}{\partial z} &= \im \frac{4 \pi \omega \chi^{\left(3\right)}}{c n_{2\omega}}\left(A_0 A_1^{2}  + \highlight{2\gamma_{12} A_1 A_1^{\ast} A_2 + \gamma_{22} A_2 A_2^{*} A_2} \right) \\
						\frac{\partial A_{0}}{\partial t} &= \alpha_2 \left|A_2\right|^2 \cdot \left(u e^{\im\left[\arg{A_2}-2\arg{A_1}\right]}-A_{0}\right)
					\end{aligned}
				\end{equation*}
			}	
			
		
		\vfill
		\small
		$A_0$, $A_1$, $A_2$ --- комплексные амплитуды квазистатического поля, поля накачки и поля второй гармоники;
		
		$c$ --- скорость света, $\omega$ --- угловая частота накачки;
		
		$n_{\omega}, n_{2\omega}$ --- показатели преломления ОВ на частотах накачки и второй гармоники;		
		
		\end{column}
		\begin{column}{0.42\textwidth}
			\small
			
			$u= 5~\text{статВ/см}\rightarrow1.5$кВ/см --- амплитуда насыщенного квазистатического поля, определяет максимальную эффективность ГВГ;
			
			$\alpha_2$ --- коэффициент, определяющий скорость насыщения; $(\alpha_2 u^{2})^{-1} = 5~\text{часов}$
			
			$\gamma_{ij}$, $i,j = \overline{1,2}$ --- коэффициенты, пропорциональные интегралам перекрытия мод. 
						
			Амплитуды полей соответствуют начальным пиковым мощностям накачки $P_1 = 4$~кВт и второй гармоники $P_2 = 500$~нВт.
		\end{column}
	\end{columns}	
\end{frame}

\begin{frame}
	\justifying 
	\framesubtitle{2. Кривые фазового синхронизма при ГВГ в оптических волокнах}
	\frametitle{Положение, выносимое на защиту}
	
	\large
	
	В процессе оптического полинга германосиликатного оптического волокна излучением ближнего ИК-диапазона ширина кривых фазового синхронизма при генерации второй гармоники монотонно уменьшается, а их пик сдвигается к расстройке, определяемой фазовой самомодуляцией и кросс-модуляцией. При этом выход эффективности генерации второй гармоники на постоянный уровень наступает до завершения эволюции кривых фазового синхронизма, поскольку уменьшение ширины кривых фазового синхронизма и смещение их пиковых значений компенсируются одновременным ростом пиковой эффективности преобразования.
\end{frame}
