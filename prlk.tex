\begin{frame}
	\frametitle{Поправки к температурным кинетикам}
	\framesubtitle{Влияние непостоянства температуры окружающей среды}
		
	\begin{columns}
		\begin{column}{0.55\textwidth}			
			\begin{itemize}
				\item Для \textbf{экспоненциальной и импульсной} методик: \fixme{что-то про Фурье}
				
				\begin{equation*}
					\epsilon(t) = \sum_{\omega_k} \dfrac{i \omega_k \hat{T_a}(\omega_k)}{\Gamma + i\omega_k} [\exp(-\Gamma t) - \exp(i \omega_k t)]
					\label{eq:correction_ta}
				\end{equation*}

				\item Для \textbf{градиентной} методики: 
				
				\begin{equation*}
					G_{\text{corr}} \coloneqq G - \left(\frac{d T_a}{d t} \bigg|_{t < t_0, T = T_0} - \frac{d T_a}{ dt} \bigg|_{t > t_0, T = T_0}\right)
					\label{eq:g_corr_ta}
				\end{equation*}
				
				\footnotesize $\epsilon$ --- поправка к $T(t)$; $\omega_k,\hat T_a(\omega_k)$ --- гармоники $T_a(t)$;
				$\Gamma$ --- скорость теплоотдачи; $t$ --- время.
				
				$G$ --- разность $dT/dt$ при $T=T_0$ (нагрев--остывание);
				$t_0$ --- выключение лазера; $T_0$ --- фикс. температура.
			\end{itemize}			
		\end{column}
		\begin{column}{0.44\textwidth}
			\includegraphics[width=\linewidth]{Dissertation/images/gradient/fourier_ambient_exp}
			\centering
			\footnotesize{Зависимость погрешности определения коэффициента поглощения $\alpha/\alpha_0$ от отношения скоростей теплоотдачи окружающей среды и кристалла $\Gamma_a/\Gamma$. Кривые 1--3: экспоненциальная, импульсная, градиентная методики с учётом поправки; кривые 4--6: методики без коррекции}
		\end{column}
	\end{columns}	
\end{frame}

	\begin{columns}
		\begin{column}{0.55\textwidth}			
			\begin{itemize}
				\item Для \textbf{экспоненциальной} методики: без изменений	
				\item Для \textbf{импульсной} методики: 
								
				\begin{equation*}
					I^\natural := \frac{C_{\mathrm{exp}} f_{\mathrm{puls}}\left( {t_0}/2 \right)}
					{\exp \left( C_{\mathrm{puls}} {t_0}/{2} \right) - \exp \left( -C_{\mathrm{exp}} {t_0}/{2} \right)},
					\label{eq:barq_puls_corr}
				\end{equation*}
				
				\item Для \textbf{градиентной} методики: 
				
				\begin{equation*}
					G_{\text{corr}} \coloneqq G - {T_0}\left( {{C_{\text{exp}}} - {C_{\text{puls}}}} \right).
				\end{equation*}
				
				\footnotesize{$f_{\mathrm{puls}}$ --- экстраполяция кинетики остывания; $C_{\mathrm{exp}},C_{\mathrm{puls}}$ --- показатели экспонент нагрева/остывания; $t_0$ --- конец нагрева.}
			\end{itemize}
