	Методики обработки температурных кинетик:
	
	\begin{columns}
		\begin{column}{0.6\textwidth}			
			\begin{itemize}
				\item \textbf{Экспоненциальный}: аппроксимация кинетики разогрева
				\item \textbf{Импульсный}: экстраполяция кинетики остывания
				\item \textbf{Градиентный}: определение коэффициента поглощения по разности производных разогрева и остывания
			\end{itemize}					
		\end{column}
		\begin{column}{0.4\textwidth}
			\includegraphics[width=\linewidth]{Dissertation/images/review/illustrative_example}
			\centering
			\footnotesize{Иллюстрация методик на примере кинетики нагрева и остывания}
		\end{column}
	\end{columns}	
	
	\vfill
	
	В отсутствие погрешностей величина теплового источника $\overline{Q}$ для всех трёх методик совпадает:
	
	$ E=I=G=\overline{Q} = \alpha LP/(mc_{\text{sp}})$
	
	\footnotesize{$\alpha$~--- коэффициент поглощения, $L, m, c_{\text{sp}}$~--- длина, масса, удельная теплоёмкость кристалла, $P$ --- мощность излучения.}

\begin{frame}
	\frametitle{Поправки к температурным кинетикам}
	\framesubtitle{Влияние непостоянства температуры окружающей среды}
	
	\fixme{Мотивация: работаем в ограниченном объеме без термостата, так называемое солнце и ночь. Кинетика Ta любая}
		
	\begin{columns}
		\begin{column}{0.55\textwidth}			
			\begin{itemize}
				\item Для \textbf{экспоненциальной и импульсной} методик: корректировка кинетики перед аппроксимацией \fixme{что-то про Фурье}
				
				\( \epsilon(t) = \sum_{\omega_k} \dfrac{\im \omega_k \hat{T_a}(\omega_k)}{\Gamma + \im\omega_k} [\exp(-\Gamma t) - \exp(\im \omega_k t)] \)

				\item Для \textbf{градиентной} методики корректировка величины
				 теплового источника: 
				
				\(
					G_{\text{corr}} \coloneqq G - \left(\dfrac{d T_a}{d t} \bigg|_{t < t_0, T = T_0} - \dfrac{d T_a}{ dt} \bigg|_{t > t_0, T = T_0}\right)
				\)
				
				\footnotesize $\epsilon$ --- поправка к $T(t)$; $\omega_k,\hat T_a(\omega_k)$ --- гармоники $T_a(t)$;
				$\Gamma$ --- скорость теплоотдачи; $t$ --- время,	$t_0$ --- выключение лазера; $T_0$ --- фикс. температура.
			\end{itemize}			
		\end{column}
		\begin{column}{0.44\textwidth}
			\includegraphics[width=\linewidth]{Dissertation/images/gradient/fourier_ambient_exp}
			\centering
			\footnotesize{Зависимость погрешности определения коэффициента поглощения $\alpha/\alpha_0$ от отношения скоростей теплоотдачи окружающей среды и кристалла $\Gamma_a/\Gamma$. 1--3: экспоненциальная, импульсная, градиентная методики с учётом поправки; 4--6: без коррекции}
		\end{column}
	\end{columns}	
\end{frame}

	\frametitle{Поправки к температурным кинетикам}
	\framesubtitle{Влияние непостоянства коэффициента теплоотдачи}
	\fixme{Мотивация: хз че там по обдуву образца если на воздуха. Пусть разные, но постоянные $h$ при нагреве и охлаждении}
	\begin{columns}
		\begin{column}{0.55\textwidth}			
			\begin{itemize}
				\item Для \textbf{экспоненциальной} методики: без изменений	
				\item Для \textbf{импульсной} методики: 
								
				\begin{equation*}
					I_{\text{corr}} := \frac{C_{\mathrm{exp}} f_{\mathrm{puls}}\left( {t_0}/2 \right)}
					{\exp \left( C_{\mathrm{puls}} {t_0}/{2} \right) - \exp \left( -C_{\mathrm{exp}} {t_0}/{2} \right)},
				\end{equation*}
				
				\item Для \textbf{градиентной} методики: 
				
				\begin{equation*}
					G_{\text{corr}} \coloneqq G - {T_0}\left( {{C_{\text{exp}}} - {C_{\text{puls}}}} \right).
				\end{equation*}
				
				\footnotesize{$f_{\mathrm{puls}}$ --- экстраполяция кинетики остывания; $C_{\mathrm{exp}},C_{\mathrm{puls}}$ --- показатели экспонент нагрева/остывания; $t_0$ --- конец нагрева.}
			\end{itemize}			
		\end{column}
		\begin{column}{0.44\textwidth}
			\includegraphics[width=\linewidth]{Dissertation/images/gradient/heat_transfer_account_var3}
			\centering
			\footnotesize{Зависимость погрешности измерения коэффициента оптического поглощения $\alpha/\alpha_0$ от относительного изменения коэффициента теплоотдачи $\Delta h / h$. 1~--- экспоненциальная методика; 2,~3~--- импульсная, градиентная методики с коррекцией; 4,~5--- импульсная, градиентная методики без коррекции}
		\end{column}
	\end{columns}
