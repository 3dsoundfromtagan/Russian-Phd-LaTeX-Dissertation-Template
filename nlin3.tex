	\begin{center}
		\centering
		\includegraphics[width=0.9\linewidth]{Presentation/images/problem_fiber}
	\end{center}

	\begin{columns}
		\begin{column}{0.35\textwidth}
			\centering
			\includegraphics[width=\linewidth]{Dissertation/images/review/fwm_principle}
			
			\footnotesize{Принципиальная схема процесса ЧВС. Из закона сохранения энергии следует, что $\omega_1+\omega_2=\omega_3+\omega_4$}
		\end{column}
		\begin{column}{0.65\textwidth}
			Условие фазового синхронизма:
			
			\vspace{-0.3cm}
			
			\begin{equation*}
				\Delta k ={{n_3}\omega_3}/{c}+{{n_4}\omega_4}/{c}-{{n_1}\omega_1}/{c}-{{n_2}\omega_2}/{c}
				\label{eq:deltaK}
			\end{equation*}
			
			${n_{i}}$~--- эффективный модовый показатель преломления, $c$~--- скорость света в вакууме.
			
			Реализация:
			
			\begin{itemize}
				\item Одномодовый (ОМ) ЧВС: аномальная дисперсия при $\lambda > 1300$~нм
				
				\item Межмодовый (ММ) ЧВС: фазовый синхронизм  за счёт разности $n_i$ в спектральной области $\lambda\sim1$~мкм.
			\end{itemize}
		\end{column}
	\end{columns}
	
	\note{Основной нежелательный эффект третьего порядка в ОВ — ЧВС: процесс, в котором пара фотонов одного спектрального и модового состава преобразуется в пару фотонов другого спектрального и модового состава. Поскольку волокно исследуемого источника поддерживает две моды на длине волны накачки (1030 нм) и одну моду на длине волны затравки (1562 нм), в нём возможно протекание как одномодового (ОМЧВС), так и межмодового ЧВС (ММЧВС) при выполнении соответствующих условий фазового синхронизма.}
	
\end{frame}
